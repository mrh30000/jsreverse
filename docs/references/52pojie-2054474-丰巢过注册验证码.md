    """

        bg_img: 滑块背景图片(文件路径或OpenCV图像对象)

    """

    box, confidence = recognizer.identify_gap(source=bg_img, verbose=False)

    logger.debug(f'可信度: {confidence}')

    # 计算滑动距离 (原始代码中的/r544可能是笔误，实际应为/)
    return int(box[0]-10)  # 将原始坐标转换为目标距离

def parse_traceData(slide_distance, pointY):
    traceData = []
    x = 0
    y = pointY
    t = int(time.time() * 1000)
    while x < slide_distance:
        trace = {"x": x, "y": y, "time": t}
        traceData.append(trace)
        x += random.randint(0, 3)
        y = pointY
        t += random.randint(1, 20)
        if x >= slide_distance:
            x = slide_distance
            trace = {"x": x, "y": y, "time": t}
            traceData.append(trace)
            break

    return traceData
