function run() {
    encrypted_url = encrypt("原视频地址")
    time = Date['now']()
    encrypted_time = encrypt(time)
    key = encrypt(sign(hex_md5(time+"原视频地址")))
    return [encrypted_url, encrypted_time, key]
}
