## QQ音乐技术分析

### 一、前言

本文分析QQ音乐Web端的音频获取机制，通过纯Python实现歌曲搜索、音质选择和音频URL获取。

核心技术发现：搜索API返回的`mid`（song\_mid）与CDN实际使用的`media_mid`是两个不同值，必须使用`media_mid`才能正确获取音频。

> 本文仅供技术学习交流，请于24小时内删除下载内容，支持正版。

### 三、Cookie获取与验证

脚本启动时自动检测本地Cookie，流程如下：

```
启动 → 检测本地 Cookie
  ├─ 存在 → 验证（搜索+获取播放URL）
  │   ├─ 通过 → 直接执行用户请求
  │   └─ 失败 → 启动浏览器重新登录
  └─ 不存在 → 启动浏览器登录
        ├─ 登录成功 → 验证 → 保存 Cookie
        └─ 登录失败 → 提示"未登录/非会员"，询问重试
```

### 四、搜索实现

搜索使用`DoSearchForQQMusicDesktop`方法，返回完整歌曲信息：

```
def search_song(keyword, cookie=""):
    """搜索歌曲，返回 [{name, mid, singer, media_mid}]"""
    payload = json.dumps({
        "comm": {"ct": 24, "cv": 0},
        "req_1": {
            "method": "DoSearchForQQMusicDesktop",
            "module": "music.search.SearchCgiService",
            "param": {
                "remoteplace": "txt.yqq.song", "search_type": 0,
                "query": keyword, "page_num": 1, "num_per_page": 10,
            }
        }
    }).encode("utf-8")
    req = urllib.request.Request(VKEY_API, data=payload,
        headers={**HEADERS, "Content-Type": "application/json", "Cookie": cookie})
    resp = urllib.request.urlopen(req, timeout=10)
    data = json.loads(resp.read().decode("utf-8"))
    songs = []
    for item in data.get("req_1", {}).get("data", {}).get("body", {}).get("song", {}).get("list", []):
        singer_list = item.get("singer", [])
        songs.append({
            "name": item.get("name", ""),
            "mid": item.get("mid", ""),
            "singer": singer_list[0].get("name", "") if singer_list else "",
            "media_mid": item.get("file", {}).get("media_mid", ""),
        })
    return songs
```

响应关键字段：

```
{
  "name": "七里香",
  "mid": "004Z8Ihr0JIu5s",
  "singer": [{"name": "周杰伦"}],
  "file": {
    "media_mid": "0012Ez0a1tFcOI",

    "size_320mp3": 10792943,

  }

**核心发现：`mid`（song\_mid）≠ `file.media_mid`（media\_mid）**

### 五、获取播放URL

使用`vkey.GetVkeyServer`模块，**必须用`media_mid`构建filename**：

```
def get_play_url(songmid, media_mid="", cookie="", quality="M800"):
    """获取播放URL"""
    ext = EXT_MAP.get(quality, ".mp3")
    file_mid = media_mid if media_mid else songmid
    filename = f"{quality}{file_mid}{ext}"
    uin_match = re.search(r"uin=(\d+)", cookie)
    uin = uin_match.group(1) if uin_match else "0"
    payload = json.dumps({
        "req_0": {
            "module": "vkey.GetVkeyServer", "method": "CgiGetVkey",
            "param": {
                "filename": [filename], "guid": "10000",
                "songmid": [songmid], "songtype": [0],
                "uin": uin, "loginflag": 1, "platform": "20",
            }
        },
        "comm": {"uin": int(uin) if uin.isdigit() else 0, "format": "json", "ct": 24, "cv": 0}
    }).encode("utf-8")
    req = urllib.request.Request(VKEY_API, data=payload,
        headers={**HEADERS, "Content-Type": "application/json", "Cookie": cookie})
    resp = urllib.request.urlopen(req, timeout=15)
    data = json.loads(resp.read().decode("utf-8"))
    vkey_data = data.get("req_0", {}).get("data", {})
    sip = vkey_data.get("sip", [])
    midurlinfo = vkey_data.get("midurlinfo", [{}])
    if not midurlinfo:
        return None, "midurlinfo 为空"
    purl = midurlinfo[0].get("purl", "")
    if not purl:
        return None, f"purl 为空 (result={midurlinfo[0].get('result', -1)})"
    if not sip:
        return None, "sip 为空"
    return sip[0] + purl, None
```

### 六、song\_mid 与 media\_mid 的区别

**问题现象：**

* 用`song_mid`构建filename → API返回`result=0`看似正常 → CDN返回404
* 用`media_mid`构建filename → CDN返回200 → 下载成功

**原因：** CDN上的音频文件使用`media_mid`作为文件名标识，与歌曲的`song_mid`是独立的两个ID。

```
# 七里香示例
song_mid  = "004Z8Ihr0JIu5s"   # 搜索返回的mid
media_mid = "0012Ez0a1tFcOI"   # file.media_mid字段

# 用song_mid → 404
filename = f"M800{song_mid}.mp3"   # M800004Z8Ihr0JIu5s.mp3 → 404

# 用media_mid → 200
filename = f"M800{media_mid}.mp3"  # M8000012Ez0a1tFcOI.mp3 → 成功
```

**注意：** vkey请求的`param.songmid`字段仍使用`song_mid`，只有`filename`使用`media_mid`。

### 七、支持的音质格式

| 参数 | 格式 | 码率 | 说明 |
| --- | --- | --- | --- |
| C200 | m4a | 48k | 低码率AAC |
| C400 | m4a | 128k | 标准AAC |
| C600 | m4a | 320k | 高码率AAC |
| M500 | mp3 | 128k | 标准MP3 |
| M800 | mp3 | 320k | 高码率MP3（推荐） |
| O400 | ogg | 128k | OGG格式 |
| O600 | ogg | 320k | 高码率OGG |
| F000 | flac | 无损 | FLAC无损 |

不可用：WAV(W000)、APE(RS01)、Hi-Res(RS02)、DSD(D000)

### 九、使用方法

```
# 搜索歌曲（首次运行会自动弹出浏览器登录）
python qq_music_dl.py --search "周杰伦"

# 搜索并下载（默认 M800=320k mp3）

# 强制重新登录
python qq_music_dl.py --login

# 手动指定Cookie
python qq_music_dl.py --cookie "uin=xxx; qqmusic_key=xxx" --play "七里香"
```

### 附录：完整脚本 qq\_music\_dl.py

```
"""
QQ Music Downloader - 一体化脚本
自动检测登录状态，未登录则启动浏览器登录，验证通过后搜索/下载

Usage:
    python qq_music_dl.py --play "歌曲名"
    python qq_music_dl.py --search "歌曲名"
    python qq_music_dl.py --play "歌曲名" --quality F000
    python qq_music_dl.py --play "歌曲名" --no-download
    python qq_music_dl.py --mid "004Z8Ihr0JIu5s"
    python qq_music_dl.py --login
    python qq_music_dl.py --cookie "uin=xxx; qqmusic_key=xxx" --play "歌曲名"
"""
import argparse
import json
import os
import re
import sys
import time
import urllib.request

try:
    from playwright.sync_api import sync_playwright
except ImportError:
    sync_playwright = None

# --- 常量 ---
VKEY_API = "https://u.y.qq.com/cgi-bin/musicu.fcg"
COOKIE_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)), ".qq_music_cookie.json")

HEADERS = {
    "Referer": "https://y.qq.com/",
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
}

EXT_MAP = {
    "C200": ".m4a", "C400": ".m4a", "C600": ".m4a",
    "M500": ".mp3", "M800": ".mp3",
    "O400": ".ogg", "O600": ".ogg",
    "F000": ".flac",
}

# ==================== Cookie 管理 ====================

def load_cookies():
    """从本地文件加载 cookie"""
    if not os.path.exists(COOKIE_FILE):
        return None
    with open(COOKIE_FILE, "r", encoding="utf-8") as f:
        data = json.load(f)
    cookies = data.get("cookies", {})
    if not cookies:
        return None
    expires_at = cookies.get("psrf_access_token_expiresAt", "")
    if expires_at and expires_at.isdigit():
        remaining = int(expires_at) - time.time()
        if remaining <= 0:
            return None
    return cookies

def save_cookies(cookies):
    """保存 cookie 到本地文件"""
    with open(COOKIE_FILE, "w", encoding="utf-8") as f:
        json.dump({
            "cookies": cookies,
            "saved_at": time.strftime("%Y-%m-%d %H:%M:%S"),
        }, f, ensure_ascii=False, indent=2)

def cookies_to_string(cookies):
    return "; ".join(f"{k}={v}" for k, v in cookies.items())

# ==================== Cookie 验证 ====================

def validate_cookie(cookie_str):
    """验证 Cookie 是否可用：搜索 + 获取播放 URL
    返回: (ok: bool, msg: str, song_info: dict|None)
    """
    # 1. 搜索
    payload = json.dumps({
        "comm": {"ct": 24, "cv": 0},
        "req_1": {
            "method": "DoSearchForQQMusicDesktop",
            "module": "music.search.SearchCgiService",
            "param": {
                "remoteplace": "txt.yqq.song", "search_type": 0,
                "query": "周杰伦", "page_num": 1, "num_per_page": 1,
            }
        }
    }).encode("utf-8")
    try:
        req = urllib.request.Request(VKEY_API, data=payload,
            headers={**HEADERS, "Content-Type": "application/json", "Cookie": cookie_str})
        resp = urllib.request.urlopen(req, timeout=10)
        data = json.loads(resp.read().decode("utf-8"))
        song_list = data.get("req_1", {}).get("data", {}).get("body", {}).get("song", {}).get("list", [])
        if not song_list:
            return False, "搜索无结果，账号可能未登录", None
    except Exception as e:
        return False, f"搜索请求失败: {e}", None

    song = song_list[0]
    songmid = song.get("mid", "")
    media_mid = song.get("file", {}).get("media_mid", "")
    if not songmid or not media_mid:
        return False, "搜索结果缺少 mid 或 media_mid", None

    # 2. 获取播放 URL
    uin_match = re.search(r"uin=(\d+)", cookie_str)
    uin = uin_match.group(1) if uin_match else "0"
    filename = f"M800{media_mid}.mp3"

    payload = json.dumps({
        "req_0": {
            "module": "vkey.GetVkeyServer", "method": "CgiGetVkey",
            "param": {
                "filename": [filename], "guid": "10000",
                "songmid": [songmid], "songtype": [0],
                "uin": uin, "loginflag": 1, "platform": "20",
            }
        },
        "comm": {"uin": int(uin) if uin.isdigit() else 0, "format": "json", "ct": 24, "cv": 0}
    }).encode("utf-8")
    try:
        req = urllib.request.Request(VKEY_API, data=payload,
            headers={**HEADERS, "Content-Type": "application/json", "Cookie": cookie_str})
        resp = urllib.request.urlopen(req, timeout=15)
        data = json.loads(resp.read().decode("utf-8"))
        vkey_data = data.get("req_0", {}).get("data", {})
        midurlinfo = vkey_data.get("midurlinfo", [{}])
        if not midurlinfo:
            return False, "midurlinfo 为空", None
        purl = midurlinfo[0].get("purl", "")
        result_code = midurlinfo[0].get("result", 0)
        if not purl:
            if result_code == 0:
                return False, "权限不足，可能不是会员", None
            return False, f"purl 为空 (result={result_code})", None
        sip = vkey_data.get("sip", [])
        if not sip:
            return False, "sip 为空", None
        return True, "验证通过", {"songmid": songmid, "media_mid": media_mid}
    except Exception as e:
        return False, f"vkey 请求失败: {e}", None

# ==================== Playwright 登录 ====================

def login_with_playwright():
    """启动浏览器登录，成功返回 cookie dict，失败返回 None"""
    if sync_playwright is None:
        print("[错误] 需要安装 playwright: pip install playwright && playwright install chromium")
        return None

    print("\n" + "=" * 50)
    print("  启动浏览器，请登录 QQ 音乐账号")
    print("=" * 50)

    while True:
        print("\n启动浏览器...")
        with sync_playwright() as p:
            browser = p.chromium.launch(headless=False, channel="chrome")
            context = browser.new_context(viewport={"width": 1280, "height": 800})
            page = context.new_page()

            input("请在浏览器中登录 QQ 音乐，完成后回到此窗口按回车...")

            page.wait_for_timeout(1000)

            cookies = {}
            for c in context.cookies():
                if "qq.com" in c.get("domain", ""):
                    cookies[c["name"]] = c["value"]

            context.close()
            browser.close()

        if not cookies:
            print("\n未获取到 Cookie，浏览器中可能未登录。")

        else:
            print(f"验证失败: {msg}")
            if input("重新登录？(y/n): ").strip().lower() == "y":
                continue
            return None

# ==================== 确保登录 ====================

def ensure_login(force=False):
    """确保有可用的 cookie，返回 cookie 字符串"""
    if not force:
        cookies = load_cookies()
        if cookies:
            cookie_str = cookies_to_string(cookies)
            print("[cookie] 检测到本地 Cookie，正在验证...")
            ok, msg, _ = validate_cookie(cookie_str)
            if ok:
                print(f"[cookie] {msg}")
                return cookie_str
            else:
                print(f"[cookie] 验证失败: {msg}，需要重新登录")

    cookies = login_with_playwright()
    if cookies:

def search_song(keyword, cookie=""):
    """搜索歌曲，返回 [{name, mid, singer, media_mid}]"""
    payload = json.dumps({
        "comm": {"ct": 24, "cv": 0},
        "req_1": {
            "method": "DoSearchForQQMusicDesktop",
            "module": "music.search.SearchCgiService",
            "param": {
                "remoteplace": "txt.yqq.song", "search_type": 0,
                "query": keyword, "page_num": 1, "num_per_page": 10,
            }
        }
    }).encode("utf-8")
    req = urllib.request.Request(VKEY_API, data=payload,
        headers={**HEADERS, "Content-Type": "application/json", "Cookie": cookie})
    resp = urllib.request.urlopen(req, timeout=10)
    data = json.loads(resp.read().decode("utf-8"))
    songs = []
    for item in data.get("req_1", {}).get("data", {}).get("body", {}).get("song", {}).get("list", []):
        singer_list = item.get("singer", [])
        songs.append({
            "name": item.get("name", ""),
            "mid": item.get("mid", ""),
            "singer": singer_list[0].get("name", "") if singer_list else "",
            "media_mid": item.get("file", {}).get("media_mid", ""),
        })
    return songs

def get_play_url(songmid, media_mid="", cookie="", quality="M800"):
    """获取播放 URL"""
    ext = EXT_MAP.get(quality, ".mp3")
    file_mid = media_mid if media_mid else songmid
    filename = f"{quality}{file_mid}{ext}"
    uin_match = re.search(r"uin=(\d+)", cookie)
    uin = uin_match.group(1) if uin_match else "0"
    payload = json.dumps({
        "req_0": {
            "module": "vkey.GetVkeyServer", "method": "CgiGetVkey",
            "param": {
                "filename": [filename], "guid": "10000",
                "songmid": [songmid], "songtype": [0],
                "uin": uin, "loginflag": 1, "platform": "20",
            }
        },
        "comm": {"uin": int(uin) if uin.isdigit() else 0, "format": "json", "ct": 24, "cv": 0}
    }).encode("utf-8")
    req = urllib.request.Request(VKEY_API, data=payload,
        headers={**HEADERS, "Content-Type": "application/json", "Cookie": cookie})
    resp = urllib.request.urlopen(req, timeout=15)
    data = json.loads(resp.read().decode("utf-8"))
    vkey_data = data.get("req_0", {}).get("data", {})
    sip = vkey_data.get("sip", [])
    midurlinfo = vkey_data.get("midurlinfo", [{}])
    if not midurlinfo:
        return None, "midurlinfo 为空"
    purl = midurlinfo[0].get("purl", "")
    if not purl:
        return None, f"purl 为空 (result={midurlinfo[0].get('result', -1)})"
    if not sip:
        return None, "sip 为空"
    return sip[0] + purl, None

def download_audio(url, output_path):
    """下载音频文件"""
    req = urllib.request.Request(url, headers={
        "Referer": "https://y.qq.com/", "User-Agent": HEADERS["User-Agent"]})
    resp = urllib.request.urlopen(req, timeout=120)
    with open(output_path, "wb") as f:
        while True:
            chunk = resp.read(8192)
            if not chunk:
                break
            f.write(chunk)
    return os.path.getsize(output_path)

def main():

    parser.add_argument("--search", help="搜索歌曲关键词")

                        choices=["C200", "C400", "C600", "M500", "M800", "O400", "O600", "F000"],

    parser.add_argument("--cookie", default="", help="手动指定 cookie 字符串")
    parser.add_argument("--output", "-o", help="输出文件路径")
    parser.add_argument("--no-download", action="store_true", help="只获取 URL，不下载")
    parser.add_argument("--login", action="store_true", help="强制重新登录")
    args = parser.parse_args()

    # 手动指定 cookie 时跳过登录流程

    else:

        if not cookie:

    # 搜索模式
    if args.search:

        if not songs:
            print("未找到结果")
            return
        for i, s in enumerate(songs):
            mm = s.get('media_mid', '')
            print(f"  [{i+1}] {s['name']} - {s['singer']} (mid: {s['mid']}, media_mid: {mm})")
        return

    song_name = "song"

        media_mid = songs[0].get("media_mid", "")

        else:
            output_path = os.path.join(os.path.dirname(os.path.abspath(__file__)), f"{safe_name}{ext}")

if __name__ == "__main__":
