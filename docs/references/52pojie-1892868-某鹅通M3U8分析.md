[JavaScript] *纯文本查看*

```
function decrypt(e) {
    var e = e.replace("__ba", "")
    var e = e.replace(/(@|#|\$|%)/g, (function(e, t){
        var n = {
            '@': '1',
            '#': '2',
            '$': '3',
            '%': '4'
        };
        return n[t]
    }))
    var buffer = Buffer.from(e, 'base64');
    var decodedString = buffer.toString('utf-8');

    return JSON.parse(decodedString)
}
```
