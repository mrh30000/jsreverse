标准的md5加密，没有魔改，非常幸运
mobile是自己输入的手机号

编写代码：

[JavaScript] *纯文本查看*

```
const cryptoJs = require('crypto-js')

function getRandom() {
    for (var e = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789", n = 6, a = "", t = 0; t < n; t++) {
        var o = Math.floor(Math.random() * e.length);
        a += e.charAt(o)
    }
    return a
}

function run(mobile){
    rw = getRandom()
    sw = cryptoJs.MD5(rw + mobile + '57fe2f9fb248428b72bf0fa33f1924d5').toString()
    return [rw,sw]
}
```

```
const cryptoJs = require('crypto-js')
function l(e) {
    var n = cryptoJs.enc.Latin1.parse("h5LoginKey123456")
        , a = cryptoJs.enc.Latin1.parse("h5LoginIv1234567")
        , t = e
        , o = cryptoJs.AES.encrypt(t, n, {
        iv: a,
        mode: cryptoJs.mode.CBC,
        padding: cryptoJs.pad.ZeroPadding
    });
    return o.toString()
}

function run(password){
    return l(password)
}
console.log(l("afdaafdadf"))
```

```
import random
import ddddocr
import requests

code_url = "https://hotel.bestwehotel.com/api/safeverify/getImageVerify"
code_params = {
    "mobile": "adfa",
    "verifyImageKey": random.random()
}
code_res = requests.get(code_url, code_params)

# 获取图片的二进制内容
image_data = code_res.content

# 将图片保存到本地
with open('test.jpg', 'wb') as f:
    f.write(image_data)

ocr = ddddocr.DdddOcr()
img_path = "test.jpg"
# 读取图片
with open(img_path, 'rb') as f:
    img_bytes = f.read()
# 识别验证码
code = ocr.classification(img_bytes)
print(code)
```
