```
def encrypt4Login(username, password):
    # 固定 Base64 密钥（来自前端）
    key_b64 = "n8KGIbDIjQvLsQvqGuzX07dUkBvtrPFWJT76ORsNun7"

    # 补全 Base64 padding
    padding = len(key_b64) % 4
    if padding:
        key_b64 += '=' * (4 - padding)

    # 解码密钥
    key = base64.b64decode(key_b64)

    # 生成 16 字节随机 IV
    iv = get_random_bytes(16)
    print(iv)

    # AES/CBC/PKCS7 加密函数
    def encrypt(plain_text):
        cipher = AES.new(key, AES.MODE_CBC, iv)
        cipher_text = cipher.encrypt(pad(plain_text.encode('utf-8'), AES.block_size))
        return base64.b64encode(cipher_text).decode('utf-8')

    # 加密用户名和密码
    return {
        "userName": encrypt(username),
        "password": encrypt(password),
        "rid": base64.b64encode(iv).decode('utf-8')  # 返回 Base64 编码的 IV
    }
```
