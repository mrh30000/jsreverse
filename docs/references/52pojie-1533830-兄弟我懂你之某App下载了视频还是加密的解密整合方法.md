基于现在网络技术的发达，很多App对视频的加密。为了视频流畅，对视频的切割。所以怎么拿到一个完整又不加密的视频还能转存任何播放器随意播放呢！！！！！
比如一个视频你可以下载：下载下来是这样的： vajhgjasgfg.ts  或者 hskghyuey.bin  几百上千个这种片段，而且还加密无法使用其他播放器打开！
1.先将下载好的视频文件夹找到：
2.找到视频加密 key
3.找到视频m3u8加密文件加密信息： 包括文件名
4.解密
5.将解密后的视频合并  得到完整视频

合并使用  TS合并工具
具体解密方法如下：

```
from Crypto.Cipher import AES
import re
import os
#ss = re.findall('^[A-Za-z0-9]+$')

def key1():
    with open('D:\\test\\www\\key.key','r') as f:
        s = f.read()
        #print(s)
        return s

def filename():
    lst1 = []
    with open('D:\\test\\www\\index.m3u8','r') as f:
        lst = f.readlines()
        for i in lst:
            #print(i)
            name = re.findall('^[A-Za-z0-9]+\.bin+$',i)
            if name:
                lst1.append(name[0])
    return lst1

#print(filename())

def decryptSingleAES( src,dec,key,iv ):
    try:
        f1 = open(src, 'rb')
        f2 = open(dec, 'wb')
        part = f1.read()
        cipher = AES.new(key.encode('utf-8'), AES.MODE_CBC, iv)
        plain_data = cipher.decrypt(part)
        if part:
            f2.write(plain_data)
    finally:
        if f1:
            f1.close()
        if f2:
            f2.close()

#if __name__ == '__main__':
os.mkdir('D:\\test\\video')
n = 1
for i in filename():
    src='D:\\test\\www\\'+i
    dec='D:\\test\\video\\'+str(n)+'.ts'
    key=str(key1())
    iv=b'0000000000000000'
    decryptSingleAES( src,dec,key,iv )
    n+=1
```
