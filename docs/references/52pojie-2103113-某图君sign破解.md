def sign_system(e: str) -> str:
    """
    函数的签名生成逻辑
    """

    timestamp = str(round(time.time()))
    e = e + "###" + timestamp

    # __APP__.wxapkg 解密 里面的 app-service.js
    t = "PXhw7UT1B0a9kQDKZsjIASmOezxYG4CHo5Jyfg2b8FLpEvRr3WtVnlqMidu6cN"
    i = ""

    for char in e:
        o = t.find(char)

        if o == -1:
            s_char = char
        else:
            s_char = t[(o + 3) % 62]

        n = int(62 * random.random())
        d = int(62 * random.random())

        i += t[n] + s_char + t[d]

    return i

def request_album_photoList(homepage_id):

    url = "https://stapi.youpengw.com/miniwechat/v1/shentujun/photoList"

    # token
    e_param = COMMON_E_PARAM

    # 初始化变量
    all_data = []
    page = 1
    limit = 30  # 这里必须和接口保持一致！！！
    max_pages = 50

    while page <= max_pages:
        try:
            # 每次重新生成时间戳 + 签名（关键！）
            import time
            timestamp = int(time.time() * 1000)
            sign = sign_system(e_param)

            params = {
                "homepage_id": homepage_id,
                "classify_id": 0,
                "sort_type": "default",
                "album_id": -2,
                "uid": 990439,
                "page": page,
                "limit": limit,  # 统一
                "id": 0,
                "action": "cover",
                "version": 1,
                "appid": "wxba7de3a24a3ff4d5",
                "openid": COMMON_openid,
                "timestamp": timestamp
            }

            # 请求头
            headers = get_common_headers(sign)

            response = requests.post(url, headers=headers, json=params, timeout=20)
            result = response.json()

            # 请求失败直接结束
            if result.get("code") != 1 or not result.get("data"):
                print(f"&#10060; 第{page}页请求失败或无数据，结束翻页")
                break

            current_page_data = result["data"].get("photoList", [])

            # 关键：当前页没有数据 → 结束翻页
            if not current_page_data:
                print(f"&#9989; 第{page}页无数据，全部爬取完毕")
                break

            all_data.extend(current_page_data)
            print(f"&#9989; 第{page}页：获取到 {len(current_page_data)} 条，累计 {len(all_data)} 条")

            page += 1
            time.sleep(0.5)

        except Exception as e:
            print(f"&#10060; 第{page}页异常：{str(e)}")
            break

    # 解析出所有的URL
    classify_id = all_data[0].get("classify_id")

    # print(classify_id)
    # print(all_data)
    # exit()

    # classify_id=1 视频  classify_id=5 图片 classify_id=7 链接   classify_id=10 集合

    if classify_id == 10:
        return classify_id

    else:

        # 解析出所有视频URL

def download_video(video_info,title):
    """
    下载单个视频
    :param video_info: 包含id和url的字典
    :param save_dir: 保存目录（自动创建）
    """
    session = requests.Session()
    session.trust_env = False
    # 禁用代理

    video_id  = video_info["id"]
    video_url = video_info["url"]

    file_ext = os.path.splitext(video_url)[1]

    # 创建保存目录
    save_dir = f"download_videos/{title}"

    # Path(save_dir).mkdir(exist_ok=True)
    Path(save_dir).mkdir(parents=True, exist_ok=True)

    # 文件名：用ID命名，避免乱码
    filename  = f"video_{video_id}{file_ext}"
    save_path = os.path.join(save_dir, filename)

    # 如果文件已存在，跳过
    if os.path.exists(save_path):
        print(f"&#8505;&#65039;  文件已存在，跳过：{filename}")
        return True

    try:
        # 流式下载（支持大文件）
        response = requests.get(
            video_url,
            stream=True,
            timeout=60,
            headers={
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/126.0.0.0 Safari/537.36"
            }
        )
        response.raise_for_status()  # 检查HTTP错误

        # 获取文件大小（显示进度）
        total_size = int(response.headers.get("Content-Length", 0))
        downloaded_size = 0

        # 写入文件
        with open(save_path, "wb") as f:
            for chunk in response.iter_content(chunk_size=8192):
                if chunk:
                    f.write(chunk)
                    downloaded_size += len(chunk)
                    # 显示下载进度
                    if total_size > 0:
                        progress = (downloaded_size / total_size) * 100
                        print(f"\r&#128229; 下载中 {filename}：{progress:.1f}% ({downloaded_size/1024/1024:.2f}MB/{total_size/1024/1024:.2f}MB)", end="")
        print(f"\n&#9989; 下载完成：{filename} → 保存路径：{save_path}")
        return True
    except Exception as e:
        print(f"\n&#10060; 下载失败 {filename}：{str(e)}")
        # 删除不完整的文件
        if os.path.exists(save_path):
            os.remove(save_path)
        return False
