     # 创建 MD5 哈希
     md5_hash = hashlib.md5(combined_string.encode()).hexdigest()
 
     # 处理哈希前八位
     def process_hash(hash_str):
         hex_value = int(hash_str[:8], 16)
         str_value = str(hex_value)[:6]
         if len(str_value) < 6:
             str_value += '0' * (6 - len(str_value))
         return str_value
 
     processed_hash = process_hash(md5_hash)
