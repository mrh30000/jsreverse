def extract_h5st_method2(text):
    # 匹配从 { 开始到 } 结束的完整对象
    pattern = r'{\s*appid:\s*[^}]+\bh5st:\s*\'([^\']+)\'[^}]*}'
    match = re.search(pattern, text, re.DOTALL)
    if match:
        return match.group(1)
    return None
