// ==UserScript==
// @name         media-source-extracter
// @namespace    [url=https://github.com/Momo707577045/media-source-extract]https://github.com/Momo707577045/media-source-extract[/url]
// @version      1.0.0
// @description  媒体缓存捕捉,会下载到一个音频和一个视频,需FFMPEG等工具手动合并
// @author       Momo707577045
// @match        https://*.cctv.com/*/V*.shtml*
// @match        https://*.cctv.cn/*/V*.shtml*
// @exclude      *://*.luckly-mjw.cn/*
// @grant        none
// @run-at       document-start
// ==/UserScript==

(function () {
  'use strict';
  localStorage.getItem('h5vodLastPostion') ? localStorage.setItem('h5vodLastPostion',"{}") : "{}";
  // 防止重复注入
  if (document.getElementById('media-source-extract')) return;

  // === StreamSaver 初始化（略作简化）===
  window.streamSaver = (function () {
    const global = window;
    if (!global.HTMLElement) console.warn('StreamSaver 需要在浏览器的主线程上运行');

    let mitmTransporter = null;
    let supportsTransferable = false;
    const ponyfill = global.WebStreamsPolyfill || {};
    const isSecureContext = global.isSecureContext;
    let useBlobFallback = /constructor/i.test(global.HTMLElement) || !!global.safari || !!global.WebKitPoint;
    const downloadStrategy = isSecureContext || 'MozAppearance' in document.documentElement.style ? 'iframe' : 'navigate';

    const streamSaver = {
      createWriteStream,
      WritableStream: global.WritableStream || ponyfill.WritableStream,
      supported: true,
      version: { full: '2.0.5', major: 2, minor: 0, dot: 5 },
      mitm: 'https://upyun.luckly-mjw.cn/lib/stream-saver-mitm.html'
    };

    function makeIframe(src) {
      const iframe = document.createElement('iframe');
      iframe.hidden = true;
      iframe.src = src;
      iframe.loaded = false;
      iframe.name = 'iframe';
      iframe.postMessage = (...args) => iframe.contentWindow.postMessage(...args);
      iframe.addEventListener('load', () => iframe.loaded = true, { once: true });
      document.body.appendChild(iframe);
      return iframe;
    }

    function makePopup(src) {
      const options = 'width=200,height=100';
      const delegate = document.createDocumentFragment();
      const popup = {
        frame: global.open(src, 'popup', options),
        loaded: false,
        isIframe: false,
        isPopup: true,
        remove() { popup.frame.close(); },
        addEventListener(...args) { delegate.addEventListener(...args); },
        dispatchEvent(...args) { delegate.dispatchEvent(...args); },
        removeEventListener(...args) { delegate.removeEventListener(...args); },
        postMessage(...args) { popup.frame.postMessage(...args); }
      };

      const onReady = evt => {
        if (evt.source === popup.frame) {
          popup.loaded = true;
          global.removeEventListener('message', onReady);
          popup.dispatchEvent(new Event('load'));
        }
      };
      global.addEventListener('message', onReady);
      return popup;
    }

    try {
      new Response(new ReadableStream());
      if (isSecureContext && !('serviceWorker' in navigator)) useBlobFallback = true;
    } catch (err) {
      useBlobFallback = true;
    }

    function test(fn) {
      try { fn(); } catch (e) { /* ignore */ }
    }

    test(() => {
      const { readable } = new TransformStream();
      const mc = new MessageChannel();
      mc.port1.postMessage(readable, [readable]);
      mc.port1.close(); mc.port2.close();
      supportsTransferable = true;
      Object.defineProperty(streamSaver, 'TransformStream', { configurable: false, writable: false, value: TransformStream });
    });

    function loadTransporter() {
      if (!mitmTransporter) {
        mitmTransporter = isSecureContext ? makeIframe(streamSaver.mitm) : makePopup(streamSaver.mitm);
      }
    }

    function createWriteStream(filename, options, size) {
      let opts = options || {};
      if (typeof options === 'number') {
        console.warn('[StreamSaver] Deprecated usage');
        opts.size = options;
      }

      let bytesWritten = 0;
      let downloadUrl = null;
      let channel = null;
      let ts = null;
      let writer = null;
      let windowNum = 0;
      let chunks = [];

      filename = encodeURIComponent(filename.replace(/\//g, ':')).replace(/['()]/g, escape).replace(/\*/g, '%2A');

      if (!useBlobFallback) {
        loadTransporter();
        channel = new MessageChannel();

        const response = {
          transferringReadable: supportsTransferable,
          pathname: opts.pathname || Math.random().toString().slice(-6) + '/' + filename,
          headers: {
            'Content-Type': 'application/octet-stream; charset=utf-8',
            'Content-Disposition': "attachment; filename*=UTF-8''" + filename
          }
        };
        if (opts.size) response.headers['Content-Length'] = opts.size;

        const args = [response, '*', [channel.port2]];

        if (supportsTransferable) {
          const transformer = downloadStrategy === 'iframe' ? undefined : {
            transform(chunk, controller) {
              if (!(chunk instanceof Uint8Array)) throw new TypeError('Can only write Uint8Arrays');
              bytesWritten += chunk.length;
              controller.enqueue(chunk);
              if (downloadUrl) {
                windowNum++;
                location.href = downloadUrl;
                downloadUrl = null;
              }
            },
            flush() {
              if (downloadUrl) {
                windowNum++;
                location.href = downloadUrl;
              }
            }
          };
          ts = new streamSaver.TransformStream(transformer, opts.writableStrategy, opts.readableStrategy);
          const readableStream = ts.readable;
          channel.port1.postMessage({ readableStream }, [readableStream]);
        }

        channel.port1.onmessage = evt => {
          if (evt.data.download) {
            if (downloadStrategy === 'navigate') {
              mitmTransporter.remove(); mitmTransporter = null;
              if (bytesWritten) {
                windowNum++; location.href = evt.data.download;
              } else {
                downloadUrl = evt.data.download;
              }
            } else {
              if (mitmTransporter.isPopup) {
                mitmTransporter.remove(); mitmTransporter = null;
                if (downloadStrategy === 'iframe') makeIframe(streamSaver.mitm);
              }
              makeIframe(evt.data.download);
            }
          } else if (evt.data.abort) {
            chunks = []; channel.port1.postMessage('abort');
            channel.port1.onmessage = null; channel.port1.close(); channel.port2.close(); channel = null;
          }
        };

        if (mitmTransporter.loaded) {
          mitmTransporter.postMessage(...args);
        } else {
          mitmTransporter.addEventListener('load', () => mitmTransporter.postMessage(...args), { once: true });
        }
      }

      writer = (!useBlobFallback && ts?.writable) || new streamSaver.WritableStream({
        write(chunk) {
          if (!(chunk instanceof Uint8Array)) throw new TypeError('Can only write Uint8Arrays');
          if (useBlobFallback) {
            chunks.push(chunk);
          } else {
            channel.port1.postMessage(chunk);
            bytesWritten += chunk.length;
            if (downloadUrl) {
              windowNum++; location.href = downloadUrl; downloadUrl = null;
            }
          }
        },
        close() {
          if (useBlobFallback) {
            const blob = new Blob(chunks, { type: 'application/octet-stream; charset=utf-8' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = filename;
            link.click();
          } else {
            channel.port1.postMessage('end');
          }
        },
        abort() {
          chunks = [];
          channel.port1.postMessage('abort');
          channel.port1.onmessage = null;
          channel.port1.close(); channel.port2.close(); channel = null;
        }
      }, opts.writableStrategy);

      const originWriter = writer.getWriter();
      writer.getWriter = () => originWriter;

      window.addEventListener('beforeunload', () => {
        if (windowNum === 0) originWriter.close();
        windowNum--;
      });

      return writer;
    }

    return streamSaver;
  })();

  // === 绕过检测 & sandbox 处理 ===
  Function.prototype.toString.call = function (caller) {
    return `'function ${caller.name}() { [native code] }'`;
  };

  setInterval(() => {
    try {
      Array.from(document.querySelectorAll('iframe[sandbox]')).forEach(iframe => {
        const parent = iframe.parentNode;
        const clone = iframe.cloneNode(true);
        clone.removeAttribute('sandbox');
        iframe.remove();
        parent.appendChild(clone);
      });
    } catch (e) { console.log(e); }
  }, 1000);

  // === 核心变量 ===
  let sumFragment = 0;
  let isStreamDownload = false;
  let _sourceBufferList = [];
  let captureFromStart = true; // 是否从头开始捕获

  // === 工具函数 ===
  function getDocumentTitle() {
    try { return window.top.document.title; } catch (e) { return document.title; }
  }

  // === DOM 元素创建 ===
  const $container = document.createElement('div');
  const $downloadNum = document.createElement('div');
  const $btnDownload = document.createElement('div');
  const $btnStreamDownload = document.createElement('div');
  const $tenRate = document.createElement('div');
  const $closeBtn = document.createElement('div');
  const $showBtn = document.createElement('div');
  const $btnCaptureFromStart = document.createElement('div');

  $btnDownload.id = 'media-source-extract';

  // Base style
  const baseStyle = `
    float: right;
    clear: both;
    margin-top: 20px;
    padding: 0 20px;
    color: white;
    cursor: pointer;
    font-size: 16px;
    font-weight: bold;
    line-height: 40px;
    text-align: center;
    border-radius: 4px;
    background-color: #3498db;
    box-shadow: 0 3px 6px 0 rgba(0, 0, 0, 0.3);
  `;

  // Icons
  $closeBtn.innerHTML = `<button style="padding-top:4px;width:24px;display:inline-block;cursor:pointer;">X</button>`;
  $showBtn.innerHTML = `<button style="padding-top:4px;width:24px;display:inline-block;cursor:pointer;">+</button>`;

  // === 功能函数 ===
  function _tenRatePlay() {
    const isTen = $tenRate.innerHTML === '十倍速捕获';
    const rate = isTen ? 10 : 1;
    $tenRate.innerHTML = isTen ? '恢复正常播放' : '十倍速捕获';
    document.querySelectorAll('video').forEach(v => v.playbackRate = rate);
  }

  function _toggleCaptureFromStart() {
    captureFromStart = !captureFromStart;
    $btnCaptureFromStart.textContent = captureFromStart ? '从头捕获：开启' : '从头捕获：关闭';
    $btnCaptureFromStart.style.backgroundColor = captureFromStart ? '#2ecc71' : '#3498db';
    localStorage.getItem('h5vodLastPostion') ? localStorage.setItem('h5vodLastPostion',"{}") : "{}";
  }

  function _download() {
    const validBuffers = _sourceBufferList.filter(target =>
      target.bufferList && target.bufferList.length > 0 && target.bytesWritten > 0
    );

    if (validBuffers.length === 0) {
      alert('没有找到有效的媒体数据，无法下载！');
      return;
    }

    console.log(`准备下载 ${validBuffers.length} 个有效缓冲区`);

    validBuffers.forEach(target => {
      const mime = target.mime.split(';')[0];
      const type = mime.split('/')[0] == "video" ? "m4v" : "m4a";
      const blob = new Blob(target.bufferList, { type: mime });
      const a = document.createElement('a');
      a.download = `${getDocumentTitle()}.${type}`;
      a.href = URL.createObjectURL(blob);
      a.style.display = 'none';
      a.onclick = e => e.stopPropagation();
      document.body.appendChild(a);
      a.click();
      a.remove();
      console.log(`已下载文件: ${getDocumentTitle()}.${type}, 大小: ${target.bytesWritten} 字节`);
    });
  }

  function _streamDownload() {
    const remain = [];
    _sourceBufferList.forEach(target => {
      const hasData = target.bufferList?.length > 0 || (target.streamWriter && target.bytesWritten > 0);
      if (target.MSEInstance.readyState === 'ended') {
        if (hasData && target.streamWriter) {
          target.streamWriter.close().catch(console.warn);
        }
        // 不再保留已结束的轨道
      } else {
        // 仍在接收数据的轨道保留
        remain.push(target);
      }
    });
    _sourceBufferList = remain;
  }

  // 自动暂停播放器
  function _autoPausePlayer() {
    // 查找页面上的所有视频元素并暂停它们
    const videos = document.querySelectorAll('video');
    videos.forEach(video => {
      if (!video.paused) {
        video.pause();
        console.log('视频已自动暂停');
      }
      localStorage.getItem('h5vodLastPostion') ? localStorage.setItem('h5vodLastPostion',"{}") : "{}";
    });
  }

  // === Hook MediaSource ===
  const _origEndOfStream = window.MediaSource.prototype.endOfStream;
  window.MediaSource.prototype.endOfStream = function () {
    if (isStreamDownload) {
      setTimeout(_autoPausePlayer, 500);
      setTimeout(_streamDownload, 100);
    } else {
      // 过滤掉空的缓冲区后再下载
      const validBuffers = _sourceBufferList.filter(target =>
        target.bufferList && target.bufferList.length > 0 && target.bytesWritten > 0
      );

      if (validBuffers.length === 0) {
        alert('没有找到有效的媒体数据，无法下载！');
      } else {
        console.log(`准备下载 ${validBuffers.length} 个有效缓冲区`);
        setTimeout(_autoPausePlayer, 500);
        _download();
      }
    }
    return _origEndOfStream.call(this);
  };

  const _origAddSourceBuffer = window.MediaSource.prototype.addSourceBuffer;
  window.MediaSource.prototype.addSourceBuffer = function (mime) {
    _appendDom(); // 确保只插入一次

    // 如果启用了从头捕获，在添加source buffer时重置播放器
    if (captureFromStart) {
      setTimeout(() => {
        const videos = document.querySelectorAll('video');
        videos.forEach(video => {
          if (video.currentTime > 0) {
            video.currentTime = 0; // 重置到开头
            console.log('已重置视频到开头以确保从头捕获');
          }
        });
      }, 500);
    }

    const sourceBuffer = _origAddSourceBuffer.call(this, mime);
    const bufferList = [];
    const entry = { mime, bufferList, MSEInstance: this, bytesWritten: 0 };
    _sourceBufferList.push(entry);

    const _origAppend = sourceBuffer.appendBuffer;
    sourceBuffer.appendBuffer = function (buffer) {
      sumFragment++;
      $downloadNum.textContent = `已捕获 ${sumFragment} 个片段`;

      // 记录该轨道的数据量
      entry.bytesWritten = (entry.bytesWritten || 0) + buffer.byteLength;

      if (isStreamDownload && entry.streamWriter) {
        entry.streamWriter.write(new Uint8Array(buffer)).catch(console.error);
      } else {
        bufferList.push(buffer);
      }
      return _origAppend.call(this, buffer);
    };

    return sourceBuffer;
  };

  window.MediaSource.prototype.addSourceBuffer.toString = () => 'function addSourceBuffer() { [native code] }';

  // === DOM 构建与事件绑定 ===
  function _appendDom() {
    if (document.getElementById('media-source-extract')) return;

    $container.id = 'media-source-extract-container';

    `;
    $showBtn.style = `
      float: right;
      clear: both;
      display: none;
      margin-top: 4px;
      height: 34px;
      width: 34px;
      line-height: 34px;
      text-align: center;
      border-radius: 4px;
      background-color: rgba(0,0,0,0.5);
    `;

    $downloadNum.textContent = '已捕获 0 个片段';
    $btnDownload.textContent = '下载已捕获片段';
    $btnStreamDownload.textContent = '特大视频下载，边下载边保存';
    $tenRate.textContent = '十倍速捕获';
    $btnCaptureFromStart.textContent = captureFromStart ? '从头捕获：开启' : '从头捕获：关闭';

    $btnDownload.addEventListener('click', _download);
    $tenRate.addEventListener('click', _tenRatePlay);
    $btnCaptureFromStart.addEventListener('click', _toggleCaptureFromStart);
    $btnStreamDownload.addEventListener('click', () => {
      isStreamDownload = true;
      $btnDownload.style.display = 'none';
      $btnStreamDownload.style.display = 'none';

      _sourceBufferList.forEach(sb => {
        // 检查是否有实际数据才初始化流写入器
        if (sb.bytesWritten > 0 && !sb.streamWriter) {
          try {
            const ext = sb.mime.split(';')[0].split('/')[1];
            sb.streamWriter = streamSaver.createWriteStream(`${getDocumentTitle()}.${ext}`).getWriter();
            // 写入已缓存的数据
            sb.bufferList?.forEach(buf => {
              sb.streamWriter.write(new Uint8Array(buf)).catch(console.error);
            });
            sb.bufferList = []; // 清空缓存
            console.log(`已为 ${ext} 类型创建流写入器，开始流式下载`);
          } catch (e) {
            console.error('流式写入器创建失败:', e);
          }
        } else if (sb.bytesWritten === 0) {
          console.log(`跳过创建流写入器，因为缓冲区为空: ${sb.mime}`);
        }
      });
    });

    $closeBtn.addEventListener('click', () => {
      [$downloadNum, $btnDownload, $btnStreamDownload, $tenRate, $btnCaptureFromStart, $closeBtn].forEach(el => el.style.display = 'none');
      $showBtn.style.display = 'inline-block';
    });

    $showBtn.addEventListener('click', () => {
      $downloadNum.style.display = 'inline-block';
      $tenRate.style.display = 'inline-block';
      $btnCaptureFromStart.style.display = 'inline-block';
      $closeBtn.style.display = 'inline-block';
      if (!isStreamDownload) {
        $btnDownload.style.display = 'inline-block';
        $btnStreamDownload.style.display = 'inline-block';
      }
      $showBtn.style.display = 'none';
    });

    document.documentElement.insertBefore($container, document.head);
    [$btnStreamDownload, $downloadNum, $btnDownload, $tenRate, $btnCaptureFromStart, $closeBtn, $showBtn].forEach(el => $container.appendChild(el));
  }
})();
