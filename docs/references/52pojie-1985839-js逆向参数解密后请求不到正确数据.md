def decrypt_data(data: str, aes_data: dict):
    """
    对获取到的数据进行解密
    :return:
    """

    AES_SECRET_KEY = aes_data["aesKey"]
    AES_SECRET_IV = aes_data["aesIv"]

    # 判断是否加密
    if not AES_SECRET_KEY or not AES_SECRET_IV:
        return data

    # 数据预处理
    data = data.rstrip("\n")

    # 将base64编码的数据解码为字节串
    encrypted_bytes = base64.b64decode(data)

    # 创建AES解密对象
    cipher = AES.new(AES_SECRET_KEY.encode("utf8"), AES.MODE_CBC, AES_SECRET_IV.encode('utf-8'))

    # 进行解密操作
    decrypted_bytes = cipher.decrypt(encrypted_bytes)

    # 去除填充（使用的是Pkcs7填充模式）
    decrypted_bytes = unpad(decrypted_bytes, AES.block_size)

    # 将解密后的字节串解码为utf8的字符串
    return decrypted_bytes.decode("utf8")
