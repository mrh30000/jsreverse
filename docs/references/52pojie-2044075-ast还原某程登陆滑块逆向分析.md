import struct

def word_array_to_hex_string(words, sig_bytes):
    byte_list = []
    for word in words:
        if word < 0:
            word = word & 0xFFFFFFFF
        bytes_ = struct.pack('>I', word)
        byte_list.extend(bytes_)
    byte_list = byte_list[:sig_bytes]

def aesEncrypt(plaintext):
